Generate missing dates wsing Recursive CTE
WITH RECURSIVE Dates AS (
SELECT MIN(order_date) AS day FROM orders
UNION ALL
SELECT day + INTERVAL 1 DAY FROM Dates

WHERE day < (SELECT MAX(order_date) FROM orders)
)
SELECT d.day, COUNT(o.order_id)
FROM Dates d
LEFT JOIN orders o ON d.day = o.order_date
GROUP BY 1;
